// api/validate.js — Vercel Serverless Function
// Valida claves de licencia contra Google Sheets CSV (server-side)
// y devuelve un token firmado que el cliente almacena en localStorage.

const crypto = require('crypto');

const CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vSyxrO896ICOc_J0N_brzrrXX64gmQ4t4uuhdrh_3_8EwumuIEQ3WaoF2oQysCtVWdM7n5rZHrrlSXG/pub?output=csv';

// Secret para firmar tokens — configurar como Variable de Entorno en Vercel
// Si no existe, usa un fallback (funcional pero menos seguro)
const SECRET = process.env.GESTO_SECRET || 'gesto-smart-default-secret-change-me';

// Rate limiting simple en memoria (se resetea entre cold starts)
const intentos = {};

function limpiarIntentos(ip) {
    const ahora = Date.now();
    if (!intentos[ip]) return;
    intentos[ip] = intentos[ip].filter(t => ahora - t < 300000); // 5 min
    if (intentos[ip].length === 0) delete intentos[ip];
}

function estaBloqueado(ip) {
    limpiarIntentos(ip);
    return intentos[ip] && intentos[ip].length >= 5;
}

function registrarIntento(ip) {
    if (!intentos[ip]) intentos[ip] = [];
    intentos[ip].push(Date.now());
}

function firmarToken(payload) {
    const header = Buffer.from(JSON.stringify({ alg: 'HS256', typ: 'JWT' })).toString('base64url');
    const body = Buffer.from(JSON.stringify(payload)).toString('base64url');
    const firma = crypto.createHmac('sha256', SECRET).update(header + '.' + body).digest('base64url');
    return header + '.' + body + '.' + firma;
}

function verificarToken(token) {
    try {
        const partes = token.split('.');
        if (partes.length !== 3) return null;
        const [header, body, firma] = partes;
        const firmaEsperada = crypto.createHmac('sha256', SECRET).update(header + '.' + body).digest('base64url');
        if (firma !== firmaEsperada) return null;
        const payload = JSON.parse(Buffer.from(body, 'base64url').toString());
        if (payload.exp && Date.now() > payload.exp) return null;
        return payload;
    } catch {
        return null;
    }
}

async function leerCSV() {
    const res = await fetch(CSV_URL);
    if (!res.ok) throw new Error('No se pudo acceder a la base de datos');
    const texto = await res.text();
    const lineas = texto.replace(/\r/g, '').split('\n');
    const clientes = {};
    for (let i = 1; i < lineas.length; i++) {
        const linea = lineas[i].trim();
        if (!linea) continue;
        const datos = linea.includes(';') ? linea.split(';') : linea.split(',');
        if (datos.length >= 3) {
            const clave = datos[0].trim().toUpperCase().replace(/^"|"$/g, '');
            const nombre = datos[1].trim().replace(/^"|"$/g, '');
            const vencimiento = datos[2].trim().replace(/^"|"$/g, '');
            if (clave && nombre) {
                clientes[clave] = { nombre, vencimiento };
            }
        }
    }
    return clientes;
}

module.exports = async function handler(req, res) {
    // CORS
    res.setHeader('Access-Control-Allow-Origin', '*');
    res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
    if (req.method === 'OPTIONS') return res.status(200).end();

    // Solo POST
    if (req.method !== 'POST') {
        return res.status(405).json({ error: 'Método no permitido' });
    }

    const ip = req.headers['x-forwarded-for'] || req.headers['x-real-ip'] || 'unknown';

    // Si es una verificación de token existente
    if (req.body.action === 'verify') {
        const payload = verificarToken(req.body.token);
        if (!payload) {
            return res.status(401).json({ error: 'Token inválido o expirado', valid: false });
        }
        return res.status(200).json({
            valid: true,
            nombre: payload.nombre,
            diasRestantes: payload.diasRestantes,
            vencimiento: payload.vencimiento
        });
    }

    // Rate limiting
    if (estaBloqueado(ip)) {
        return res.status(429).json({
            error: 'Demasiados intentos. Esperá 5 minutos e intentá de nuevo.',
            retryAfter: 300
        });
    }

    const { clave } = req.body;
    if (!clave || typeof clave !== 'string') {
        return res.status(400).json({ error: 'Falta la clave' });
    }

    try {
        const clientes = await leerCSV();
        const claveNormalizada = clave.trim().toUpperCase();
        const cliente = clientes[claveNormalizada];

        if (!cliente) {
            registrarIntento(ip);
            return res.status(401).json({ error: 'Clave incorrecta o inexistente.' });
        }

        // Validar vencimiento
        const hoy = new Date();
        hoy.setHours(0, 0, 0, 0);
        const partesFecha = cliente.vencimiento.split('-');
        if (partesFecha.length !== 3) {
            return res.status(500).json({ error: 'Error en formato de fecha' });
        }
        const fechaVenc = new Date(partesFecha[0], partesFecha[1] - 1, partesFecha[2]);
        fechaVenc.setHours(0, 0, 0, 0);
        const diasRestantes = Math.ceil((fechaVenc.getTime() - hoy.getTime()) / (1000 * 3600 * 24));

        if (diasRestantes < 0) {
            registrarIntento(ip);
            return res.status(401).json({ error: 'Clave vencida. Renová tu suscripción.' });
        }

        // Generar token (expira en 30 días o cuando venza la licencia)
        const expDias = Math.min(diasRestantes, 30);
        const token = firmarToken({
            nombre: cliente.nombre,
            vencimiento: cliente.vencimiento,
            diasRestantes,
            exp: Date.now() + (expDias * 24 * 60 * 60 * 1000),
            iat: Date.now()
        });

        return res.status(200).json({
            valid: true,
            token,
            nombre: cliente.nombre,
            diasRestantes,
            vencimiento: cliente.vencimiento
        });

    } catch (error) {
        console.error('Error en validación:', error);
        return res.status(500).json({ error: 'Error al conectar con la base de datos. Intentá de nuevo.' });
    }
};
